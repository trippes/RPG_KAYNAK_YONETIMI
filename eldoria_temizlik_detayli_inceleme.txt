   9: class QuestState(Enum): AVAILABLE="available"; ACTIVE="active"; COMPLETED="completed"; FAILED="failed"
  10: 
  11: @dataclass
  12: class Quest:
  13:     quest_id: str
  14:     title: str
  15:     objectives: tuple[Objective, ...]
  16:     reward: Reward = Reward()
  17:     prerequisites: AllConditions = AllConditions()
  18:     state: QuestState = QuestState.AVAILABLE
  19:     def __post_init__(self):
  20:         if not self.quest_id or not self.title: raise ValueError("Quest ID and title are required.")
  21:         self.state = QuestState(self.state)
  22:         ids = [o.objective_id for o in self.objectives]
  23:         if len(ids) != len(set(ids)): raise ValueError("Objective IDs must be unique.")
  24:     def can_start(self, context): return self.state == QuestState.AVAILABLE and self.prerequisites.evaluate(context)
  25:     def start(self, context=None):
  26:         if not self.can_start(context or {}): raise ValueError("Quest prerequisites are not met or quest is not available.")
  27:         self.state = QuestState.ACTIVE
  28:     def update(self, objective_id, amount=1):
  29:         if self.state != QuestState.ACTIVE: raise ValueError("Only active quests can be updated.")
  30:         objective = next((o for o in self.objectives if o.objective_id == objective_id), None)
  31:         if objective is None: raise KeyError(objective_id)
  32:         objective.advance(amount)
  33:         if all(o.completed for o in self.objectives if not o.optional): self.state = QuestState.COMPLETED
  34:         return self.state
  35:     def fail(self):
  36:         if self.state == QuestState.COMPLETED: raise ValueError("Completed quest cannot fail.")
  37:         self.state = QuestState.FAILED

  59: def load_demo_content(path: str | Path | None = None) -> dict:
  60:     """Load and validate a JSON content definition without mutating it."""
  61:     if path is None:
  62:         candidates = (
  63:             DEFAULT_CONTENT_PATH,
  64:             Path(sys.prefix) / "data" / "demo_content.json",
  65:             Path(sys.prefix) / "demo_content.json",
  66:         )
  67:         source = next((candidate for candidate in candidates if candidate.exists()), candidates[0])
  68:     else:
  69:         source = Path(path)
  70:     try:
  71:         definition = json.loads(source.read_text(encoding="utf-8"))
  72:     except (OSError, json.JSONDecodeError) as error:
  73:         raise ContentDefinitionError(f"Content definition could not be loaded: {source}") from error
  74:     if not isinstance(definition, dict):
  75:         raise ContentDefinitionError("Content definition must be a JSON object.")
