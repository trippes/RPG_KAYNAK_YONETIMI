 139:     def collect(self, node_id: str, inventory_id: str, quantity: Quantity) -> None:
 140:         """Atomically extract from an active-world node into a registered inventory."""
 141:         bundle = self._current_bundle()
 142:         world = bundle.session.worlds.get_active()
 143:         if world is None:
 144:             raise GameplayServiceError("An active world is required for collection.")
 145:         inventory = self._inventory(bundle, inventory_id)
 146:         node = bundle.resource_nodes.get(node_id)
 147:         if node is None:
 148:             raise KeyError(f"Resource node not found: {node_id}")
 149:         if node.world_id != world.world_id:
 150:             raise GameplayServiceError(
 151:                 f"Resource node {node_id} does not belong to the active world."
 152:             )
 153:         bundle.resource_nodes.extract(node_id, inventory, quantity)

 159:     def _complete_timed_action(self, action_type: ActionType, owner_id: str, callback):
 160:         """Run an immediate gameplay operation through the canonical action runtime."""
 161:         bundle = self._current_bundle()
 162:         duration = action_tick_cost(action_type)
 163:         action = bundle.simulation.actions.create(
 164:             Identifier(owner_id),
 165:             duration,
 166:             context={"action_type": action_type.value},
 167:         )
 168:         bundle.simulation.actions.schedule(
 169:             action.action_id,
 170:             bundle.simulation.current_tick,
 171:         )
 172:         bundle.production_jobs.advance_by(duration)
 173:         if action.state.value != "completed":
 174:             raise GameplayServiceError(
 175:                 f"Timed action did not complete: {action.action_id}"
 176:             )
 177:         return callback()

 195:     def start_chain(
 196:         self,
 197:         chain_id: str,
 198:         stages: Iterable[GameplayChainStage | tuple[str, str, str]],
 199:     ) -> ProductionChain:
 200:         """Start an explicit-facility sequential chain using existing jobs."""
 201:         bundle = self._current_bundle()
 202:         self._require_active_world(bundle)
 203:         if not isinstance(chain_id, str) or not chain_id.strip():
 204:             raise ValueError("chain_id cannot be empty.")
 205:         if chain_id in self._chains:
 206:             raise ValueError(f"Gameplay chain already exists: {chain_id}")
 207: 
 208:         requests = tuple(self._normalize_stage(stage) for stage in stages)
 209:         if not requests:
 210:             raise ValueError("A gameplay chain must contain at least one stage.")
 211:         normalized = tuple(
 212:             ProductionChainStage(
 213:                 recipe=self._recipe(bundle, request.recipe_id),
 214:                 inventory=self._inventory(bundle, request.inventory_id),
 215:                 facility_id=request.facility_id,
 216:             )
 217:             for request in requests
 218:         )
 219:         chain = ProductionChain(
 220:             bundle.production_jobs,
 221:             normalized,
 222:             chain_id=chain_id,
 223:             facilities=bundle.facilities,
 224:         )
 225:         try:
 226:             chain.start()
 227:         except Exception:
 228:             chain.dispose()
 229:             raise
 230:         self._chains[chain_id] = chain
 231:         return chain

 255:     def _rebind_content_quests(self) -> None:
 256:         """Reconnect optional content aliases to the restored canonical quest log."""
 257:         quest_log = self._current_bundle().quest_log
 258:         for attribute in ("available_forest_quest", "carpenter_quest", "inn_quest", "blacksmith_quest"):
 259:             current = getattr(self, attribute, None)
 260:             quest_id = getattr(current, "quest_id", None)
 261:             if quest_id is None:
 262:                 continue
 263:             restored = quest_log.get(quest_id)
 264:             if restored is not None:
 265:                 setattr(self, attribute, restored)
 266:                 if attribute in ("carpenter_quest", "inn_quest"):
 267:                     self._quest_reward_accounts.setdefault(quest_id, "hero")
 268:                     self._quest_reward_inventories.setdefault(quest_id, "player")

 270:     def register_actor(self, actor_id: str, role: str, world_id: str | None = None, active: bool = True, character_id: str | None = None) -> ActorView:
 271:         bundle = self._current_bundle()
 272:         target_world = world_id
 273:         if target_world is None:
 274:             world = bundle.session.worlds.get_active()
 275:             if world is None:
 276:                 raise GameplayServiceError("An active world is required to register an actor.")
 277:             target_world = world.world_id
 278:         character = None
 279:         if character_id is not None:
 280:             character = bundle.characters.get(character_id)
 281:             if character is None:
 282:                 raise KeyError(f"Character not found: {character_id}")
 283:             if character.actor_id is not None and character.actor_id != actor_id:
 284:                 raise ValueError(
 285:                     f"Character already references another actor: {character_id}"
 286:                 )
 287:         actor = WorldActor(actor_id, target_world, role, active)
 288:         bundle.actors.add(actor)
 289:         if character is not None:
 290:             bundle.actors.bind_character(actor_id, character)
 291:         bundle.validate()
 292:         return actor.view()

 376:     def equip_item(self, actor_id: str, inventory_id: str, item_id: str):
 377:         """Equip one catalog item, returning the replaced item to the inventory."""
 378:         bundle = self._current_bundle()
 379:         character = self.actor_character(actor_id)
 380:         if character is None:
 381:             raise GameplayServiceError("Actor must be linked to a character.")
 382:         item = bundle.items.get(item_id)
 383:         if item is None:
 384:             raise KeyError(f"Item not found: {item_id}")
 385:         inventory = self._inventory(bundle, inventory_id)
 386:         if inventory.item_quantity(item_id) < 1:
 387:             raise ValueError(f"Item is not in inventory: {item_id}")
 388: 
 389:         def equip():
 390:             previous = character.equip(item)
 391:             inventory.remove_item(item_id)
 392:             if previous is not None:
 393:                 inventory.add_item(previous)
 394:             character.current_hit_points = min(
 395:                 character.max_hit_points,
 396:                 character.current_hit_points,
 397:             )
 398:             return character.equipment.snapshot()
 399: 
 400:         return self._complete_timed_action(ActionType.INTERACT, actor_id, equip)
 401: 
 402:     def unequip_item(self, actor_id: str, inventory_id: str, slot: EquipmentSlot | str):
 403:         bundle = self._current_bundle()

 406:             raise GameplayServiceError("Actor must be linked to a character.")

 445:     def interact_with_location(self, actor_id: str, location_id: str) -> LocationInteractionView:
 446:         """Inspect a location from an actor currently standing there."""
 447:         bundle = self._current_bundle()
 448:         self._require_actor_in_active_world(bundle, actor_id)
 449:         actor = bundle.actors.get(actor_id)
 450:         world_map = bundle.movement.map_for(actor.world_id)
 451:         if world_map is None:
 452:             raise GameplayServiceError(f"No movement map registered for world: {actor.world_id}")
 453:         location = world_map.get(location_id)
 454:         if location is None:
 455:             raise KeyError(f"Location not found: {location_id}")
 456:         if actor.location_id != location_id:
 457:             raise GameplayServiceError(
 458:                 f"Actor {actor_id} is not at location: {location_id}"
 459:             )
 460:         resource_node_ids = tuple(
 461:             node.node_id for node in bundle.resource_nodes.for_world(actor.world_id)
 462:             if node.location_id is None or node.location_id == location_id
 463:         )
 464:         actor_ids = tuple(
 465:             other.actor_id
 466:             for other in bundle.actors.for_world(actor.world_id)
 467:             if other.location_id == location_id
 468:         )
 469:         return self._complete_timed_action(
 470:             ActionType.INTERACT,
 471:             actor_id,
 472:             lambda: LocationInteractionView(
 473:                 actor_id, location.location_id, location.name, actor.character_id,
 474:                 resource_node_ids, actor_ids,
 475:             ),
 476:         )

 528:     def consume_item(self, actor_id: str, inventory_id: str, resource_id: str) -> int:
 529:         """Consume one item and apply its supported character effect."""
 530:         bundle = self._current_bundle()
 531:         character = self.actor_character(actor_id)
 532:         if character is None:
 533:             raise GameplayServiceError("Actor must be linked to a character to consume items.")
 534:         resource = ResourceType(resource_id)
 535:         effects = {"bread": 2, "water": 1}
 536:         healing = effects.get(resource.value)
 537:         if healing is None:
 538:             raise GameplayServiceError(f"Resource is not consumable: {resource_id}")
 539:         inventory = self._inventory(bundle, inventory_id)
 540:         return self._complete_timed_action(
 541:             ActionType.CONSUME_ITEM,
 542:             actor_id,
 543:             lambda: self._consume_and_apply(
 544:                 inventory, resource, character, healing,
 545:             ),
 546:         )

 548:     @staticmethod
 549:     def _consume_and_apply(inventory, resource, character, healing: int) -> int:
 550:         inventory.remove(resource, Quantity(1, ResourceUnit.ITEM))
 551:         character.current_hit_points = min(
 552:             character.max_hit_points,
 553:             character.current_hit_points + healing,
 554:         )
 555:         return character.current_hit_points
